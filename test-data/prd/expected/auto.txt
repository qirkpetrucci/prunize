// Document synchronization engine
interface DocumentState {
  id: string;
  version: number;
  content: JSONContent;
  participants: Participant[];
  operations: Operation[];
}

interface Operation {
  id: string;
  type: 'insert' | 'delete' | 'format' | 'move';
  position: number;
  content?: string;
  attributes?: Record<string, any>;
  userId: string;
  timestamp: number;
  vectorClock: VectorClock;
}

class DocumentSync {
  private state: DocumentState;
  private socket: WebSocket;
  private operationQueue: Operation[] = [];
  
  constructor(documentId: string) {
    this.state = this.loadInitialState(documentId);
    this.socket = this.connectWebSocket();
    this.setupEventHandlers();
  }
  
  async applyOperation(op: Operation): Promise<void> {
    // Validate operation
    if (!this.validateOperation(op)) {
      throw new Error('Invalid operation');
    }
    
    // Transform against concurrent operations
    const transformedOp = this.transformOperation(op);
    
    // Apply to local state
    this.state = this.applyToState(this.state, transformedOp);
    
    // Broadcast to other participants
    await this.broadcastOperation(transformedOp);
    
    // Update version
    this.state.version++;
  }
  
  private transformOperation(op: Operation): Operation {
    // Operational Transformation algorithm
    const concurrentOps = this.operationQueue.filter(
      queuedOp => queuedOp.timestamp > op.timestamp
    );
    
    let transformed = op;
    for (const concurrentOp of concurrentOps) {
      transformed = this.transformPair(transformed, concurrentOp);
    }
    
    return transformed;
  }
  
  private transformPair(op1: Operation, op2: Operation): Operation {
    // IT (Inclusion Transformation) for concurrent operations
    if (op1.type === 'insert' && op2.type === 'insert') {
      if (op1.position < op2.position) {
        return op1;
      } else if (op1.position > op2.position) {
        return { ...op1, position: op1.position + (op2.content?.length || 0) };
      } else {
        // Tie-breaking using user ID
        return op1.userId < op2.userId ? op1 : { ...op1, position: op1.position + 1 };
      }
    }
    
    if (op1.type === 'delete' && op2.type === 'insert') {
      if (op1.position < op2.position) {
        return { ...op2, position: op2.position - 1 };
      }
      return op2;
    }
    
    // Handle other operation type combinations
    return op1;
  }
}

// Video call management using WebRTC
class VideoCall {
  constructor(roomId, userId) {
    this.roomId = roomId;
    this.userId = userId;
    this.localStream = null;
    this.peerConnections = new Map();
    this.signalingServer = new WebSocket('wss://signal.example.com');
    this.iceServers = [
      { urls: 'stun:stun.l.google.com:19302' },
      { 
        urls: 'turn:turn.example.com:3478',
        username: 'user',
        credential: 'pass'
      }
    ];
  }
  
  async initialize() {
    try {
      // Get user media
      this.localStream = await navigator.mediaDevices.getUserMedia({
        video: {
          width: { ideal: 1280 },
          height: { ideal: 720 },
          frameRate: { ideal: 30 }
        },
        audio: {
          echoCancellation: true,
          noiseSuppression: true,
          autoGainControl: true
        }
      });
      
      // Setup signaling handlers
      this.setupSignalingHandlers();
      
      // Join room
      this.signalingServer.send(JSON.stringify({
        type: 'join',
        roomId: this.roomId,
        userId: this.userId
      }));
      
      return this.localStream;
    } catch (error) {
      console.error('Failed to initialize video call:', error);
      throw error;
    }
  }
  
  setupSignalingHandlers() {
    this.signalingServer.onmessage = async (event) => {
      const message = JSON.parse(event.data);
      
      switch (message.type) {
        case 'user-joined':
          await this.handleUserJoined(message.userId);
          break;
        case 'offer':
          await this.handleOffer(message.userId, message.offer);
          break;
        case 'answer':
          await this.handleAnswer(message.userId, message.answer);
          break;
        case 'ice-candidate':
          await this.handleIceCandidate(message.userId, message.candidate);
          break;
        case 'user-left':
          this.handleUserLeft(message.userId);
          break;
      }
    };
  }
  
  async handleUserJoined(remoteUserId) {
    // Create peer connection
    const pc = new RTCPeerConnection({ iceServers: this.iceServers });
    this.peerConnections.set(remoteUserId, pc);
    
    // Add local tracks
    this.localStream.getTracks().forEach(track => {
      pc.addTrack(track, this.localStream);
    });
    
    // Handle ICE candidates
    pc.onicecandidate = (event) => {
      if (event.candidate) {
        this.signalingServer.send(JSON.stringify({
          type: 'ice-candidate',
          targetUserId: remoteUserId,
          candidate: event.candidate
        }));
      }
    };
    
    // Handle remote tracks
    pc.ontrack = (event) => {
      this.emit('remote-stream', {
        userId: remoteUserId,
        stream: event.streams[0]
      });
    };
    
    // Create and send offer
    const offer = await pc.createOffer();
    await pc.setLocalDescription(offer);
    
    this.signalingServer.send(JSON.stringify({
      type: 'offer',
      targetUserId: remoteUserId,
      offer: offer
    }));
  }
  
  async handleOffer(remoteUserId, offer) {
    const pc = new RTCPeerConnection({ iceServers: this.iceServers });
    this.peerConnections.set(remoteUserId, pc);
    
    this.localStream.getTracks().forEach(track => {
      pc.addTrack(track, this.localStream);
    });
    
    pc.onicecandidate = (event) => {
      if (event.candidate) {
        this.signalingServer.send(JSON.stringify({
          type: 'ice-candidate',
          targetUserId: remoteUserId,
          candidate: event.candidate
        }));
      }
    };
    
    pc.ontrack = (event) => {
      this.emit('remote-stream', {
        userId: remoteUserId,
        stream: event.streams[0]
      });
    };
    
    await pc.setRemoteDescription(new RTCSessionDescription(offer));
    const answer = await pc.createAnswer();
    await pc.setLocalDescription(answer);
    
    this.signalingServer.send(JSON.stringify({
      type: 'answer',
      targetUserId: remoteUserId,
      answer: answer
    }));
  }
  
  async toggleVideo() {
    const videoTrack = this.localStream.getVideoTracks()[0];
    videoTrack.enabled = !videoTrack.enabled;
    return videoTrack.enabled;
  }
  
  async toggleAudio() {
    const audioTrack = this.localStream.getAudioTracks()[0];
    audioTrack.enabled = !audioTrack.enabled;
    return audioTrack.enabled;
  }
  
  async startScreenShare() {
    try {
      const screenStream = await navigator.mediaDevices.getDisplayMedia({
        video: { cursor: 'always' },
        audio: false
      });
      
      const screenTrack = screenStream.getVideoTracks()[0];
      
      // Replace video track in all peer connections
      this.peerConnections.forEach((pc) => {
        const sender = pc.getSenders().find(s => s.track?.kind === 'video');
        if (sender) {
          sender.replaceTrack(screenTrack);
        }
      });
      
      // Handle screen share stop
      screenTrack.onended = () => {
        this.stopScreenShare();
      };
      
      return screenStream;
    } catch (error) {
      console.error('Screen share failed:', error);
      throw error;
    }
  }
}

from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from typing import Dict, Set
import json
import asyncio

app = FastAPI()

class ConnectionManager:
    def __init__(self):
        self.rooms: Dict[str, Set[WebSocket]] = {}
        self.user_to_room: Dict[WebSocket, str] = {}
    
    async def connect(self, websocket: WebSocket, room_id: str, user_id: str):
        await websocket.accept()
        
        if room_id not in self.rooms:
            self.rooms[room_id] = set()
        
        # Notify existing users
        await self.broadcast_to_room(room_id, {
            "type": "user-joined",
            "userId": user_id
        }, exclude=websocket)
        
        # Add to room
        self.rooms[room_id].add(websocket)
        self.user_to_room[websocket] = room_id
    
    async def disconnect(self, websocket: WebSocket, user_id: str):
        room_id = self.user_to_room.get(websocket)
        if room_id and room_id in self.rooms:
            self.rooms[room_id].discard(websocket)
            
            await self.broadcast_to_room(room_id, {
                "type": "user-left",
                "userId": user_id
            })
            
            # Clean up empty rooms
            if not self.rooms[room_id]:
                del self.rooms[room_id]
        
        if websocket in self.user_to_room:
            del self.user_to_room[websocket]
    
    async def send_to_user(self, target: WebSocket, message: dict):
        try:
            await target.send_text(json.dumps(message))
        except Exception as e:
            print(f"Error sending message: {e}")
    
    async def broadcast_to_room(self, room_id: str, message: dict, exclude: WebSocket = None):
        if room_id in self.rooms:
            tasks = []
            for connection in self.rooms[room_id]:
                if connection != exclude:
                    tasks.append(self.send_to_user(connection, message))
            await asyncio.gather(*tasks, return_exceptions=True)

manager = ConnectionManager()

@app.websocket("/ws/video/{room_id}/{user_id}")
async def video_websocket(websocket: WebSocket, room_id: str, user_id: str):
    await manager.connect(websocket, room_id, user_id)
    
    try:
        while True:
            data = await websocket.receive_text()
            message = json.loads(data)
            
            # Route signaling messages
            if message["type"] in ["offer", "answer", "ice-candidate"]:
                target_user_id = message.get("targetUserId")
                if target_user_id and room_id in manager.rooms:
                    # Find target connection
                    for conn in manager.rooms[room_id]:
                        if conn != websocket:  # Simple routing, improve with user tracking
                            await manager.send_to_user(conn, {
                                "type": message["type"],
                                "userId": user_id,
                                "offer": message.get("offer"),
                                "answer": message.get("answer"),
                                "candidate": message.get("candidate")
                            })
    
    except WebSocketDisconnect:
        await manager.disconnect(websocket, user_id)

// AI assistant service
import OpenAI from 'openai';
import { Document } from '@langchain/core/documents';
import { RecursiveCharacterTextSplitter } from 'langchain/text_splitter';

export class AIAssistant {
  private openai: OpenAI;
  private embeddingCache: Map<string, number[]>;
  
  constructor(apiKey: string) {
    this.openai = new OpenAI({ apiKey });
    this.embeddingCache = new Map();
  }
  
  async summarizeDocument(content: string, maxLength: number = 500): Promise<string> {
    const splitter = new RecursiveCharacterTextSplitter({
      chunkSize: 4000,
      chunkOverlap: 200,
    });
    
    const docs = await splitter.createDocuments([content]);
    
    // Summarize each chunk
    const chunkSummaries = await Promise.all(
      docs.map(doc => this.summarizeChunk(doc.pageContent))
    );
    
    // Combine summaries
    const combinedSummary = chunkSummaries.join('\n\n');
    
    // Final summary
    if (combinedSummary.length > maxLength * 3) {
      return this.summarizeChunk(combinedSummary, maxLength);
    }
    
    return combinedSummary;
  }
  
  private async summarizeChunk(text: string, maxTokens: number = 500): Promise<string> {
    const response = await this.openai.chat.completions.create({
      model: 'gpt-4-turbo',
      messages: [
        {
          role: 'system',
          content: 'You are a helpful assistant that creates concise, accurate summaries. Focus on key points and actionable insights.'
        },
        {
          role: 'user',
          content: `Summarize the following text:\n\n${text}`
        }
      ],
      temperature: 0.3,
      max_tokens: maxTokens,
    });
    
    return response.choices[0].message.content || '';
  }
  
  async extractActionItems(transcript: string): Promise<ActionItem[]> {
    const response = await this.openai.chat.completions.create({
      model: 'gpt-4-turbo',
      messages: [
        {
          role: 'system',
          content: `Extract action items from meeting transcripts. Return a JSON array with objects containing:
- task: brief description
- assignee: person mentioned (or null)
- dueDate: any mentioned date (or null)
- priority: high/medium/low based on urgency indicators
- context: relevant surrounding text`
        },
        {
          role: 'user',
          content: transcript
        }
      ],
      temperature: 0.2,
      response_format: { type: 'json_object' },
    });
    
    const result = JSON.parse(response.choices[0].message.content || '{}');
    return result.actionItems || [];
  }
  
  async semanticSearch(query: string, documents: Document[]): Promise<SearchResult[]> {
    // Generate query embedding
    const queryEmbedding = await this.generateEmbedding(query);
    
    // Generate document embeddings (with caching)
    const docEmbeddings = await Promise.all(
      documents.map(async (doc) => {
        const cacheKey = this.getCacheKey(doc.pageContent);
        let embedding = this.embeddingCache.get(cacheKey);
        
        if (!embedding) {
          embedding = await this.generateEmbedding(doc.pageContent);
          this.embeddingCache.set(cacheKey, embedding);
        }
        
        return { doc, embedding };
      })
    );
    
    // Calculate cosine similarity
    const results = docEmbeddings
      .map(({ doc, embedding }) => ({
        document: doc,
        score: this.cosineSimilarity(queryEmbedding, embedding),
      }))
      .sort((a, b) => b.score - a.score)
      .slice(0, 20);
    
    // Rerank with cross-encoder
    const reranked = await this.rerank(query, results);
    
    return reranked;
  }
  
  private async generateEmbedding(text: string): Promise<number[]> {
    const response = await this.openai.embeddings.create({
      model: 'text-embedding-3-large',
      input: text.slice(0, 8000), // Limit to 8K tokens
    });
    
    return response.data[0].embedding;
  }
  
  private cosineSimilarity(a: number[], b: number[]): number {
    let dotProduct = 0;
    let normA = 0;
    let normB = 0;
    
    for (let i = 0; i < a.length; i++) {
      dotProduct += a[i] * b[i];
      normA += a[i] * a[i];
      normB += b[i] * b[i];
    }
    
    return dotProduct / (Math.sqrt(normA) * Math.sqrt(normB));
  }
  
  private getCacheKey(content: string): string {
    // Simple hash function
    let hash = 0;
    for (let i = 0; i < content.length; i++) {
      const char = content.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return hash.toString(36);
  }
}
